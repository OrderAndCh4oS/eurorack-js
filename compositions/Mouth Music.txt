// MOUTH MUSIC — 84 BPM, one repeating eight-step vocal phrase.
// Continuous airflow; pitch and vowel posture glide together on one clock.
// Rhythm comes from pressure accents, without replacing the physical vocal.
// M articulation closes into a nasal consonant, then releases into each vowel.
// SLEW: 20 ms pitch / 50 ms mouth. PRESSURE SLEW: 30 ms breath accents.
// Performance: ATTN Att1 sets accent depth; VOCAL Vowel shifts the mouth;
// BREATHS Len2 changes accent length. Keep Len1 above the clock interval.

const vocalPool = Object.fromEntries('ABCDEFGHIJKLMNOPQRSTUVWXYZ'.split('').concat(['SH', 'CH', 'TH', 'DH', 'NG', 'ZH']).map(label => ['enable' + label, label === 'M' ? 1 : 0]));

const p = patch()
  .module('clk', 'tempo', { rate: 0.289432 }, { row: 1, index: 0 })
  .module('seq', 'phrase', {
    length: 8, range: 0, direction: 0,
    // E E G F# / E E A F#; the second half answers the first.
    step1: 0, step2: 0, step3: 0.25, step4: 0.166667,
    step5: 0, step6: 0, step7: 0.416667, step8: 0.166667,
    gate1: 1, gate2: 0, gate3: 1, gate4: 0,
    gate5: 1, gate6: 1, gate7: 1, gate8: 0
  }, { row: 1, index: 1 })
  .module('seq', 'vowels', {
    length: 8, range: 2, direction: 0,
    // Rounded O/U -> O -> open A -> E, with a brighter ending.
    step1: 1, step2: 0.875, step3: 0.0625, step4: 0.25,
    step5: 1, step6: 0.875, step7: 0.0625, step8: 0.5
  }, { row: 1, index: 2 })
  .module('slew', 'glide', { rate1: 0.01, rate2: 0.025 }, { row: 1, index: 3 })
  .module('gate-delay', 'breaths', {
    delay1: 0, length1: 0.7, delay2: 0, length2: 0.4533
  }, { row: 1, index: 4 })
  .module('atten', 'pressure', { atten1: 0.54, offset1: 0.5, atten2: 0.75 }, { row: 1, index: 5 })
  .module('slew', 'breathShape', { rate1: 0.015, rate2: 0.01 }, { row: 1, index: 6 })
  .module('vocal', 'vocal', {
    pitch: 7, vowel: 0.2, sound: 0, ...vocalPool, articulation: 0, pressure: 6,
    breath: 0.03, tension: 0.6, fry: 0, level: 0.8
  }, { row: 1, index: 7 })
  .module('verb', 'room', { time: 0.25, damp: 0.65, mix: 0.08 }, { row: 1, index: 8 })
  .module('out', 'output', { volume: 0.32 }, { row: 1, index: 9 })
  .connect('tempo.clock', 'phrase.clock')
  .connect('tempo.clock', 'vowels.clock')
  // This retriggered long gate keeps airflow uninterrupted between steps.
  .connect('tempo.clock', 'breaths.trig1')
  .connect('breaths.gate1', 'vocal.gate')
  .connect('phrase.cv', 'glide.in1')
  .connect('vowels.cv', 'glide.in2')
  .connect('glide.out1', 'vocal.vOct')
  .connect('glide.out2', 'vocal.vowelCV')
  // Short pressure pulses add rhythmic emphasis above the held breath.
  .connect('phrase.gate', 'breaths.trig2')
  .connect('breaths.gate2', 'pressure.in1')
  .connect('pressure.out1', 'breathShape.in1')
  .connect('breathShape.out1', 'vocal.pressureCV')
  .connect('breaths.gate2', 'pressure.in2')
  .connect('pressure.out2', 'breathShape.in2')
  .connect('breathShape.out2', 'vocal.articulationCV')
  .connect('vocal.out', 'room.audioL')
  .connect('vocal.out', 'room.audioR')
  .connect('room.outL', 'output.L')
  .connect('room.outR', 'output.R')
