// MOUTH MUSIC — 96 BPM, A minor pentatonic.
// Archived arrangement, adapted to independent BEATBOX and VOCAL modules.
// BREATHS gates VOCAL; BEATBOX receives its own trigger pattern.
// The sequencers realign every 56 eighth notes (17.5 seconds).
// Performance: BREATHS Len1 changes syllable length; LFO Rate changes vowels;
// VOCAL Fry adds rasp; DELAY Mix brings the offbeat answer forward.

const vocalPool = Object.fromEntries('ABCDEFGHIJKLMNOPQRSTUVWXYZ'.split('').concat(['SH', 'CH', 'TH', 'DH', 'NG', 'ZH']).map(label => ['enable' + label, label === 'M' ? 1 : 0]));

const p = patch()
  .module('clk', 'tempo', { rate: 0.30103 }, { row: 1, index: 0 })
  .module('swing', 'groove', { swing: 0.18, human: 0, width: 0.08, template: 0 }, { row: 1, index: 1 })
  .module('seq', 'beats', {
    length: 8, range: 2, direction: 0,
    // B, t, Pf, rest, B, tongue click, Hum, lip pop.
    step1: 0.025, step2: 0.125, step3: 0.325, step4: 0.125,
    step5: 0.025, step6: 0.625, step7: 0.925, step8: 0.725,
    gate1: 1, gate2: 1, gate3: 1, gate4: 0,
    gate5: 1, gate6: 1, gate7: 1, gate8: 1
  }, { row: 1, index: 2 })
  .module('mix', 'soundRange', { lvl1: 1, lvl2: 0.25, lvl3: 0, lvl4: 0 }, { row: 1, index: 3 })
  .module('seq', 'phrase', {
    length: 7, range: 0, direction: 0,
    step1: 0, step2: 0.25, step3: 0.583333, step4: 0.833333,
    step5: 0.583333, step6: 0.416667, step7: 0.25,
    gate1: 0, gate2: 0, gate3: 1, gate4: 0,
    gate5: 1, gate6: 1, gate7: 0, gate8: 0
  }, { row: 1, index: 4 })
  .module('gate-delay', 'breaths', { delay1: 0, length1: 0.56 }, { row: 1, index: 5 })
  .module('lfo', 'mouth', { rateKnob: 0.09, waveKnob: 0, range: 0 }, { row: 1, index: 6 })
  .module('beatbox', 'beatbox', {
    pitch: 0, shape: 0, sound: 0, pressure: 8, breath: 0.12, level: 0.8
  }, { row: 1, index: 7 })
  .module('vocal', 'vocal', {
    pitch: 0, vowel: 0, sound: 0, ...vocalPool, articulation: 0, pressure: 8,
    breath: 0.12, tension: 0.48, fry: 0.06, level: 0.8
  }, { row: 1, index: 8 })
  .module('mix', 'voices', { lvl1: 0.6, lvl2: 0.6, lvl3: 0, lvl4: 0 }, { row: 2, index: 0 })
  .module('dly', 'answer', { time: 0.234375, feedback: 0.23, mix: 0.12 }, { row: 2, index: 1 })
  .module('verb', 'room', { time: 0.35, damp: 0.6, mix: 0.16 }, { row: 2, index: 2 })
  .module('out', 'output', { volume: 0.7 }, { row: 2, index: 3 })
  .connect('tempo.clock', 'groove.clock')
  .connect('groove.swung', 'beats.clock')
  .connect('groove.swung', 'phrase.clock')
  // A 1.25x CV sum expands SEQ's 0–4 V range to all ten sound bins.
  .connect('beats.cv', 'soundRange.in1')
  .connect('beats.cv', 'soundRange.in2')
  .connect('soundRange.out', 'beatbox.soundCV')
  .connect('beats.gate', 'beatbox.trig')
  .connect('phrase.cv', 'vocal.vOct')
  .connect('phrase.gate', 'breaths.trig1')
  .connect('breaths.gate1', 'vocal.gate')
  .connect('mouth.primary', 'vocal.vowelCV')
  .connect('mouth.secondary', 'beatbox.accent')
  .connect('beatbox.out', 'voices.in1')
  .connect('vocal.out', 'voices.in2')
  .connect('voices.out', 'answer.audio')
  .connect('answer.out', 'room.audioL')
  .connect('answer.out', 'room.audioR')
  .connect('room.outL', 'output.L')
  .connect('room.outR', 'output.R')
